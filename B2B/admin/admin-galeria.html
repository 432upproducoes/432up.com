export default { async fetch(request, env) { const corsHeaders = { "Access-Control-Allow-Origin": "https://www.432up.com", "Access-Control-Allow-Methods": "PUT, OPTIONS", "Access-Control-Allow-Headers": "Authorization, Content-Type", "Access-Control-Max-Age": "86400" }; if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders }); } if (request.method !== "PUT") { return new Response("Method Not Allowed", { status: 405, headers: corsHeaders }); } try { const auth = request.headers.get("Authorization"); if (!auth || !auth.startsWith("Bearer ")) { return new Response("Unauthorized", { status: 401, headers: corsHeaders }); } const accessToken = auth.slice(7).trim(); if (!accessToken) { return new Response("Unauthorized", { status: 401, headers: corsHeaders }); } /* * 1. Valida o token diretamente no Supabase Auth. */ const userResponse = await fetch( `${env.SUPABASE_URL}/auth/v1/user`, { method: "GET", headers: { "Authorization": `Bearer ${accessToken}`, "apikey": env.SUPABASE_ANON_KEY } } ); if (!userResponse.ok) { return new Response("Unauthorized", { status: 401, headers: corsHeaders }); } const user = await userResponse.json(); if (!user || !user.id) { return new Response("Unauthorized", { status: 401, headers: corsHeaders }); } /* * 2. Consulta o parceiro correspondente ao usuário. * CORREÇÃO: a tabela "parceiros" não tem coluna "is_admin". * As colunas reais de cargo são "role" e "nivel". */ const parceiroResponse = await fetch( `${env.SUPABASE_URL}/rest/v1/parceiros` + `?select=role,nivel,status` + `&user_id=eq.${encodeURIComponent(user.id)}` + `&limit=1`, { method: "GET", headers: { "Authorization": `Bearer ${accessToken}`, "apikey": env.SUPABASE_ANON_KEY } } ); if (!parceiroResponse.ok) { return new Response("Forbidden", { status: 403, headers: corsHeaders }); } const parceiros = await parceiroResponse.json(); const parceiro = parceiros && parceiros[0]; if (!parceiro) { return new Response("Forbidden", { status: 403, headers: corsHeaders }); } /* * 3. Somente admin/master pode enviar. * CORREÇÃO: checar "role" E "nivel" (mesma lógica do HTML do painel). */ const role = String( parceiro.role || "" ).toLowerCase().trim(); const nivel = String( parceiro.nivel || "" ).toLowerCase().trim(); const isAdmin = role === "admin" || role === "master" || nivel === "admin" || nivel === "master"; if (!isAdmin) { return new Response("Forbidden", { status: 403, headers: corsHeaders }); } if ( parceiro.status === "bloqueado" || parceiro.status === "recusado" ) { return new Response("Forbidden", { status: 403, headers: corsHeaders }); } /* * 4. Caminho do arquivo. */ const url = new URL(request.url); const path = url.searchParams.get("path"); if (!path) { return new Response("Missing path", { status: 400, headers: corsHeaders }); } /* * Impede caminhos estranhos. */ if ( path.startsWith("/") || path.includes("..") || !/^(full|midias|marcas)\/[^/]+$/.test(path) ) { return new Response("Invalid path", { status: 400, headers: corsHeaders }); } /* * 5. Grava diretamente no R2. * Usamos request.body como stream para evitar * carregar o arquivo inteiro desnecessariamente * em memória. */ const contentType = request.headers.get("Content-Type") || "application/octet-stream"; const object = await env.BUCKET.put( path, request.body, { httpMetadata: { contentType: contentType, cacheControl: "public, max-age=31536000, immutable" } } ); if (!object) { return new Response("Upload failed", { status: 500, headers: corsHeaders }); } return new Response( JSON.stringify({ ok: true, path: path, url: `https://media.432up.com/${path}`, size: object.size, etag: object.etag }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } catch (error) { console.error("R2 upload error:", error); return new Response( JSON.stringify({ ok: false, error: error && error.message ? error.message : "Upload failed" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } } };
